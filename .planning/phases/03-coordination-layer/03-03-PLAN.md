```kotlin
package org.bidon.sdk.ads.cache.denis.orchestration

import org.bidon.sdk.adapter.Adapter
import org.bidon.sdk.adapter.AdaptersSource
import org.bidon.sdk.auction.models.AdUnit
import org.bidon.sdk.logs.logging.impl.logInfo

/**
 * Result of splitting waterfall into RTB and CPM groups.
 */
internal data class SplitWaterfall(
    val rtbAdUnits: List<AdUnit>,
    val cpmAdUnits: List<AdUnit>
)

/**
 * Splits auction waterfall into RTB and CPM groups based on adapter type.
 *
 * Decision logic (from 03-CONTEXT.md):
 * - Check if adapter implements Adapter.Bidding interface
 * - If Adapter.Bidding: treat as RTB only (ignore CPM config even if present)
 * - Otherwise: treat as CPM
 *
 * This matches GetTokensUseCaseImpl pattern of using filterIsInstance<Adapter.Bidding>.
 */
internal object WaterfallSplitter {
    private const val TAG = "WaterfallSplitter"

    /**
     * Split AdUnits into RTB and CPM groups.
     *
     * @param adUnits List of ad units from auction response
     * @param adaptersSource Source of available adapters
     * @return SplitWaterfall with separate RTB and CPM lists
     */
    fun split(adUnits: List<AdUnit>, adaptersSource: AdaptersSource): SplitWaterfall {
        // Build set of bidding adapter demand IDs
        // Pattern matches GetTokensUseCaseImpl: filterIsInstance<Adapter.Bidding>()
        val biddingDemandIds = adaptersSource.adapters
            .filterIsInstance<Adapter.Bidding>()
            .map { it.demandId.demandId }
            .toSet()

        // Partition AdUnits based on adapter type
        val rtbAdUnits = mutableListOf<AdUnit>()
        val cpmAdUnits = mutableListOf<AdUnit>()

        adUnits.forEach { adUnit ->
            if (adUnit.demandId in biddingDemandIds) {
                rtbAdUnits.add(adUnit)
            } else {
                cpmAdUnits.add(adUnit)
            }
        }

        logInfo(TAG, "Waterfall split: ${rtbAdUnits.size} RTB, ${cpmAdUnits.size} CPM from ${adUnits.size} total")

        return SplitWaterfall(
            rtbAdUnits = rtbAdUnits,
            cpmAdUnits = cpmAdUnits
        )
    }
}
```

Key points:
- Uses `filterIsInstance<Adapter.Bidding>()` pattern from GetTokensUseCaseImpl
- User decision: "Determine RTB vs CPM by checking Adapter.Bidding interface"
- User decision: "If adapter implements Adapter.Bidding, treat as RTB only"
- Logs split counts for debugging

```kotlin
package org.bidon.sdk.ads.cache.denis.orchestration

import org.bidon.sdk.adapter.AdaptersSource
import org.bidon.sdk.adapter.DemandAd
import org.bidon.sdk.ads.AuctionInfo
import org.bidon.sdk.ads.cache.denis.processors.CpmProcessor
import org.bidon.sdk.ads.cache.denis.processors.RtbProcessor
import org.bidon.sdk.ads.cache.denis.stores.CacheEntry
import org.bidon.sdk.ads.cache.denis.stores.ReadyToShowCache
import org.bidon.sdk.ads.cache.denis.stores.RtbPayloadCache
import org.bidon.sdk.auction.AdTypeParam
import org.bidon.sdk.auction.models.AuctionResponse
import org.bidon.sdk.auction.models.AuctionResult
import org.bidon.sdk.auction.models.TokenInfo
import org.bidon.sdk.auction.usecases.GetAuctionRequestUseCase
import org.bidon.sdk.auction.usecases.GetTokensUseCase
import org.bidon.sdk.config.BidonError
import org.bidon.sdk.logs.logging.impl.logInfo

/**
 * Coordination layer for auction orchestration.
 *
 * Acts as the "brain" between AdCache.cache() and parallel processors:
 * - Determines auction start state (warm/cold) based on cache contents
 * - Calculates dynamic pricefloor with safety margin
 * - Orchestrates token collection, auction request, and parallel processing
 *
 * Key decisions honored (from 03-CONTEXT.md):
 * - Warm start: immediate callback, no auction (serve cached ad)
 * - RTB_PAYLOAD cache: skip token collection for cached adapters
 * - Pricefloor: 0.9 safety margin on max cached eCPM
 */
internal class CoordinationLayer(
    private val adaptersSource: AdaptersSource,
    private val getTokens: GetTokensUseCase,
    private val getAuctionRequest: GetAuctionRequestUseCase,
    private val rtbProcessor: RtbProcessor,
    private val cpmProcessor: CpmProcessor,
) {
    /**
     * Determine auction start state based on cache contents.
     *
     * Decision logic:
     * 1. If READY_TO_SHOW cache not empty -> WarmStart (immediate callback)
     * 2. If RTB_PAYLOAD cache not empty -> ColdStartWithCache (skip tokens for cached)
     * 3. Otherwise -> PureColdStart (full auction)
     *
     * Cache state captured ONCE at call time (no re-validation during processing).
     */
    fun determineStartState(userPricefloor: Double): Pair<AuctionStartState, CacheStateSnapshot> {
        // Capture cache state BEFORE any async operations
        val snapshot = CacheStateSnapshot.capture()

        val state = when {
            !snapshot.readyToShowIsEmpty -> {
                // Warm start: serve cached ad immediately
                val bestAd = ReadyToShowCache.getBest()
                if (bestAd != null) {
                    logInfo(TAG, "Warm start: cached ad available (demandId=${bestAd.demandId}, ecpm=${bestAd.ecpm})")
                    AuctionStartState.WarmStart(bestAd)
                } else {
                    // Edge case: isEmpty() returned false but getBest() null (race condition)
                    logInfo(TAG, "Warning: cache reported non-empty but getBest() returned null")
                    AuctionStartState.PureColdStart(userPricefloor)
                }
            }
            !snapshot.rtbPayloadIsEmpty -> {
                // Cold start with RTB cache optimization
                logInfo(TAG, "Cold start with cache: ${snapshot.cachedDemandIds.size} RTB payloads cached")
                AuctionStartState.ColdStartWithCache(
                    cachedDemandIds = snapshot.cachedDemandIds,
                    maxCachedEcpm = snapshot.rtbPayloadMaxEcpm
                )
            }
            else -> {
                // Pure cold start
                logInfo(TAG, "Pure cold start: both caches empty")
                AuctionStartState.PureColdStart(userPricefloor)
            }
        }

        return state to snapshot
    }

    /**
     * Calculate dynamic pricefloor for auction request.
     */
    fun calculatePricefloor(userPricefloor: Double, snapshot: CacheStateSnapshot): Double {
        return PricefloorCalculator.calculateDynamicPricefloor(
            userPricefloor = userPricefloor,
            readyToShowMaxEcpm = snapshot.readyToShowMaxEcpm,
            rtbPayloadMaxEcpm = snapshot.rtbPayloadMaxEcpm
        )
    }

    /**
     * Orchestrate complete auction flow.
     *
     * Flow:
     * 1. Determine start state (warm/cold)
     * 2. If warm start: fire callback immediately, return
     * 3. If cold start: collect tokens (with skip optimization), request auction, parallel process
     *
     * @param adTypeParam Ad type parameters
     * @param demandAd Demand ad configuration
     * @param tokenTimeout Timeout for token collection
     * @param onSuccess Callback when first ad loads
     * @param onFailure Callback when auction fails
     */
    suspend fun coordinateAuction(
        adTypeParam: AdTypeParam,
        demandAd: DemandAd,
        tokenTimeout: Long,
        onSuccess: (AuctionResult, AuctionInfo) -> Unit,
        onFailure: (AuctionInfo?, BidonError) -> Unit,
    ) {
        val userPricefloor = adTypeParam.pricefloor

        // Step 1: Determine start state and capture cache snapshot
        val (startState, snapshot) = determineStartState(userPricefloor)

        when (startState) {
            is AuctionStartState.WarmStart -> {
                // Warm start: immediate callback, no auction
                handleWarmStart(startState.bestAd, onSuccess)
            }
            is AuctionStartState.ColdStartWithCache -> {
                // Cold start with RTB cache: skip tokens for cached adapters
                handleColdStart(
                    skipDemandIds = startState.cachedDemandIds,
                    snapshot = snapshot,
                    adTypeParam = adTypeParam,
                    demandAd = demandAd,
                    tokenTimeout = tokenTimeout,
                    onSuccess = onSuccess,
                    onFailure = onFailure,
                )
            }
            is AuctionStartState.PureColdStart -> {
                // Pure cold start: full token collection
                handleColdStart(
                    skipDemandIds = emptySet(),
                    snapshot = snapshot,
                    adTypeParam = adTypeParam,
                    demandAd = demandAd,
                    tokenTimeout = tokenTimeout,
                    onSuccess = onSuccess,
                    onFailure = onFailure,
                )
            }
        }
    }

    /**
     * Handle warm start: immediate callback with cached ad.
     *
     * Per 03-CONTEXT.md: "No background refresh on warm start (serve cached ad only)"
     */
    private fun handleWarmStart(
        bestAd: CacheEntry<AuctionResult>,
        onSuccess: (AuctionResult, AuctionInfo) -> Unit,
    ) {
        logInfo(TAG, "Warm start: serving cached ad immediately (demandId=${bestAd.demandId}, ecpm=${bestAd.ecpm})")

        // Create minimal AuctionInfo from cached entry
        val auctionInfo = AuctionInfo(
            auctionId = bestAd.auctionId,
            auctionConfigurationId = null, // Not available from cache
            auctionConfigurationUid = null,
            auctionTimeout = 0L, // No auction performed
            auctionPricefloor = bestAd.ecpm,
            noBids = null,
            adUnits = null,
        )

        // Fire callback immediately (synchronous, no suspend)
        onSuccess(bestAd.value, auctionInfo)
    }

    /**
     * Handle cold start: token collection, auction request, parallel processing.
     */
    private suspend fun handleColdStart(
        skipDemandIds: Set<String>,
        snapshot: CacheStateSnapshot,
        adTypeParam: AdTypeParam,
        demandAd: DemandAd,
        tokenTimeout: Long,
        onSuccess: (AuctionResult, AuctionInfo) -> Unit,
        onFailure: (AuctionInfo?, BidonError) -> Unit,
    ) {
        val dynamicPricefloor = calculatePricefloor(adTypeParam.pricefloor, snapshot)

        logInfo(TAG, "Cold start: pricefloor=${dynamicPricefloor}, skipDemandIds=${skipDemandIds.size}")

        // Step 1: Collect tokens (with skip optimization)
        val tokens = getTokens(
            adTypeParam = adTypeParam,
            adaptersSource = adaptersSource,
            tokenTimeout = tokenTimeout,
            skipDemandIds = skipDemandIds,
        )

        // Step 2: Request auction
        val auctionId = java.util.UUID.randomUUID().toString()
        val auctionResult = getAuctionRequest.request(
            adTypeParam = adTypeParam,
            auctionId = auctionId,
            demandAd = demandAd,
            adapters = adaptersSource.adapters.associate {
                it.demandId.demandId to it.adapterInfo
            },
            tokens = tokens,
        )

        auctionResult.fold(
            onSuccess = { auctionResponse ->
                handleAuctionResponse(
                    auctionResponse = auctionResponse,
                    adTypeParam = adTypeParam,
                    demandAd = demandAd,
                    dynamicPricefloor = dynamicPricefloor,
                    onSuccess = onSuccess,
                    onFailure = onFailure,
                )
            },
            onFailure = { error ->
                logInfo(TAG, "Auction request failed: ${error.message}")
                onFailure(null, BidonError.Unspecified(error))
            }
        )
    }

    /**
     * Handle auction response: split waterfall and execute parallel processing.
     */
    private suspend fun handleAuctionResponse(
        auctionResponse: AuctionResponse,
        adTypeParam: AdTypeParam,
        demandAd: DemandAd,
        dynamicPricefloor: Double,
        onSuccess: (AuctionResult, AuctionInfo) -> Unit,
        onFailure: (AuctionInfo?, BidonError) -> Unit,
    ) {
        val adUnits = auctionResponse.adUnits ?: emptyList()

        if (adUnits.isEmpty()) {
            logInfo(TAG, "Auction response has no adUnits")
            val auctionInfo = createAuctionInfo(auctionResponse)
            onFailure(auctionInfo, BidonError.NoFill(org.bidon.sdk.adapter.DemandId("auction")))
            return
        }

        // Step 3: Split waterfall into RTB and CPM groups
        val splitWaterfall = WaterfallSplitter.split(adUnits, adaptersSource)

        logInfo(TAG, "Waterfall split: RTB=${splitWaterfall.rtbAdUnits.size}, CPM=${splitWaterfall.cpmAdUnits.size}")

        // Step 4: Create callback coordinator and orchestrator
        val callbackCoordinator = CallbackCoordinator(
            onAdLoaded = onSuccess,
            onAdLoadFailed = { info, error -> onFailure(info, error) }
        )

        val orchestrator = ParallelAuctionOrchestrator(
            rtbProcessor = rtbProcessor,
            cpmProcessor = cpmProcessor,
            callbackCoordinator = callbackCoordinator,
        )

        val auctionInfo = createAuctionInfo(auctionResponse)

        // Step 5: Execute parallel auction
        orchestrator.executeParallelAuction(
            rtbPayloadsAvailable = !RtbPayloadCache.isEmpty(),
            cpmAdUnits = splitWaterfall.cpmAdUnits,
            adTypeParam = adTypeParam,
            demandAd = demandAd,
            auctionId = auctionResponse.auctionId,
            auctionConfigurationId = auctionResponse.auctionConfigurationId ?: 0L,
            auctionConfigurationUid = auctionResponse.auctionConfigurationUid ?: "",
            externalWinNotificationsEnabled = auctionResponse.externalWinNotificationsEnabled,
            pricefloor = dynamicPricefloor,
            auctionInfo = auctionInfo,
        )
    }

    /**
     * Create AuctionInfo from auction response.
     */
    private fun createAuctionInfo(response: AuctionResponse): AuctionInfo {
        return AuctionInfo(
            auctionId = response.auctionId,
            auctionConfigurationId = response.auctionConfigurationId,
            auctionConfigurationUid = response.auctionConfigurationUid,
            auctionTimeout = response.auctionTimeout,
            auctionPricefloor = response.pricefloor,
            noBids = null, // TODO: populate from response.noBids
            adUnits = null, // TODO: populate from stats
        )
    }

    companion object {
        private const val TAG = "CoordinationLayer"
    }
}
```

4. **Waterfall splitting**: Uses WaterfallSplitter.split() before parallel processing
5. **Orchestrator creation**: Creates fresh CallbackCoordinator and ParallelAuctionOrchestrator per auction
6. **AuctionInfo**: Created from auction response for callbacks
